cmake_minimum_required(VERSION 3.16)
project(ps3buzz_phase1 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(buzz_core core/buzz.c core/websocket.c core/usb_model.c core/mailbox.c kernel/engine.c)
target_include_directories(buzz_core PUBLIC core kernel)
if(MSVC)
  target_compile_options(buzz_core PRIVATE /W4 /WX)
else()
  target_compile_options(buzz_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE buzz_core)
enable_testing()
add_test(NAME core COMMAND test_core)
add_library(buzz_cell_adapter game/cellusbd_adapter.c game/import_hook.c)
target_include_directories(buzz_cell_adapter PUBLIC game)
target_link_libraries(buzz_cell_adapter PUBLIC buzz_core)
add_executable(test_game_sequence tests/test_game_sequence.c)
target_link_libraries(test_game_sequence PRIVATE buzz_cell_adapter)
add_test(NAME game_sequence COMMAND test_game_sequence)
add_executable(pc_server tests/pc_server.c net/server.c)
target_include_directories(pc_server PRIVATE net)
target_compile_definitions(pc_server PRIVATE _CRT_SECURE_NO_WARNINGS _POSIX_C_SOURCE=200809L)
target_link_libraries(pc_server PRIVATE buzz_core)
if(WIN32)
  target_link_libraries(pc_server PRIVATE ws2_32)
endif()
